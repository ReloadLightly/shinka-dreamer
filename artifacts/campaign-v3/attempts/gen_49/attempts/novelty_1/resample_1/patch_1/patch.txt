 def _routes(m, start, risk=None, reverse=False, safe=None):
     distances = {start: 0.0}
     queue = [(0.0, start)]
     opened = m["door_open"] or m["keys"] >= 2
     while queue:
         cost, p = heapq.heappop(queue)
         if cost != distances[p]:
             continue
         for d in MOVES:
             q = _add(p, d)
             origin, destination = (q, p) if reverse else (p, q)
             direction = (
                 destination[0] - origin[0],
                 destination[1] - origin[1],
             )
             if (_solid(m, q, opened)
                     or not _legal(m, origin, direction, opened)):
                 continue
             danger = 0.0 if risk is None else risk.get(destination, 0.0)
             if safe is not None:
                 if destination in m["enemies"] or danger > safe:
                     continue
             edge = 1.0
             if risk is not None:
                 edge += (
                     6.0 * danger
                     + .025 * min(12, m["visits"].get(destination, 0))

+def _collision_loss(before, after, totals, poisson, q, stay):
+    following_totals = []
+    loss = 0.0
+    for distribution, projected, old_total, is_poisson in zip(
+            before, after, totals, poisson):
+        removed = (
+            distribution.get(q, 0.0)
+            + projected.get(q, 0.0)
+            - distribution.get(q, 0.0) * stay
+        )
+        removed = min(old_total, max(0.0, removed))
+        new_total = max(0.0, old_total - removed)
+        if is_poisson:
+            # Thinning a Poisson field conditions on no collision without
+            # redistributing killed mass back into the surviving field.
+            loss += removed
+        else:
+            if new_total <= 1e-12:
+                return None
+            loss -= math.log(new_total / max(1e-12, old_total))
+        following_totals.append(new_total)
+    return loss, following_totals
