def directional_information(terms):
    # Under a uniform reference law, compute each anonymous source's
    # posterior attempted direction. Fully censored sources contribute zero.
    # Marginals are computed separately so opposing moves do not cancel.
    sources = len(terms[0])
    posterior = [[0.0] * 9 for _ in range(sources)]
    evidence = 0.0
    for term in terms:
        sizes = [mask.bit_count() for mask in term]
        weight = 1.0
        for size in sizes:
            weight *= size / 9.0
        evidence += weight
        for row, mask, size in zip(posterior, term, sizes):
            if not size:
                continue
            contribution = weight / size
            for k in range(9):
                if mask & (1 << k):
                    row[k] += contribution
    if evidence <= 1e-15:
        return 0.0
    information = 0.0
    for row in posterior:
        for value in row:
            probability = value / evidence
            if probability > 1e-15:
                information += probability * math.log(9.0 * probability)
    return min(float(sources), max(0.0, information / math.log(9.0)))
