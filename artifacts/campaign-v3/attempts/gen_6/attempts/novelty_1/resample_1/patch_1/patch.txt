 def _anonymous_evidence(theta, groups, wanted):
     """Exact likelihood and expected attempted-direction counts.
     Each source contributes a destination bit, or zero for a destination
     outside the monitored cells. OR combines anonymous occupancy, allowing
     multiple enemies to occupy the same observed cell.
     """
     options = []
     informative = []
     for source_groups in groups:
         opts = []
         for bit, directions in source_groups:
             if bit & ~wanted:
                 continue
             probability = sum(theta[d] for d in directions)
             if probability > 0:
                 opts.append((bit, directions, probability))
         if not opts:
             return 0.0, None
         options.append(opts)
         informative.append(len(source_groups) > 1)
     total = 0.0
     counts = [0.0] * 9
     chosen = []
     def visit(i, mask, weight):
         nonlocal total
         if i == len(options):
             if mask != wanted:
                 return
             total += weight
             for j, (_, directions, probability) in enumerate(chosen):
                 if informative[j]:
                     factor = weight / probability
                     for direction in directions:
                         counts[direction] += factor * theta[direction]
             return
         for option in options[i]:
             chosen.append(option)
             visit(i + 1, mask | option[0], weight * option[2])
             chosen.pop()
     visit(0, 0, 1.0)
     if total <= 1e-15:
         return total, None
     return total, [value / total for value in counts]

 def _reachable(m, risk):
