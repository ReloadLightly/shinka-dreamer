+def edge_survival(layers, fog, transition, t, p, q):
+    """One-step survival conditioned on being alive at p at time t.
+    Moving into q can kill before enemy movement. Remaining at p cannot
+    collide with an enemy already at p, because the agent is alive there.
+    The same enemy's before/after occupancy is not treated as independent.
+    """
+    survival = 1.0
+    p_to_q = transition(p).get(q, 0.0)
+    q_to_q = transition(q).get(q, 0.0)
+    for before, after in zip(layers[t], layers[t + 1]):
+        at_p = min(1.0, before.get(p, 0.0))
+        denominator = 1.0 - at_p
+        if denominator <= 1e-10:
+            return 0.0
+        if p == q:
+            danger = after.get(q, 0.0) - at_p * p_to_q
+        else:
+            at_q = before.get(q, 0.0)
+            danger = (
+                at_q + after.get(q, 0.0)
