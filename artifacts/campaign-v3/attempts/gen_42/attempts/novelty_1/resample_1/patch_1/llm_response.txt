<DESCRIPTION>
The current agent uses slow/fast learning decays of .992/.88, likelihood tempering of .8, four routing horizons, and two-step local control. This version retains the recovery expert and adds censored-observation learning, information-weighted updates, and three-step survival-conditioned control. Intermediate targets receive a completion reward once, followed by safety checks and a bounded onward-route potential; only escape is terminal. Local risk weights become 92, .58, and .27 across the three steps, with a three-candidate second-step beam. These changes aim to reduce collisions around keys and frontiers while preserving progress. Performance remains to be measured.

"""Anonymous dynamics learning with survival-conditioned target continuation."""
# EVOLVE-BLOCK-START
import heapq
import math

MOVES = tuple((x, y) for y in (-1, 0, 1) for x in (-1, 0, 1))
WALK = tuple(d for d in MOVES if d != (0, 0))
CARDINAL = ((1, 0), (-1, 0), (0, 1), (0, -1))
UNIFORM = [1.0 / 9.0] * 9
ROUTE_HORIZON = 4
SECOND_BEAM = 3


def add(p, d):
    return p[0] + d[0], p[1] + d[1]


def distance(p, q):
    return max(abs(p[0] - q[0]), abs(p[1] - q[1]))


def clamp(p):
    return min(1.0, max(0.0, p))


def blocked(terrain, p, opened=False, unknown=False):
    cell = terrain.get(p)
    if cell is None:
        return unknown
    return cell in (-1, 1) or (cell == 3 and not opened)


def destination(terrain, p, d, opened):
    q = add(p, d)
    if blocked(terrain, q, opened):
        return p
    if d[0] and d[1]:
        if (blocked(terrain, add(p, (d[0], 0)), opened)
                or blocked(terrain, add(p, (0, d[1])), opened)):
            return p
    return q


def opening_now(m):
    return m["door_open"] or (
        m["keys"] >= 2
        and any(m["map"].get(add(m["pos"], d)) == 3 for d in CARDINAL)
    )


def count_law(counts):
    total = sum(counts) + 6.3
    return [(c + .7) / total for c in counts]


def component_laws(m):
    return [UNIFORM] + [count_law(c) for c in m["counts"]]


def learned_law(m):
    laws = component_laws(m)
    return [
        sum(w * law[k] for w, law in zip(m["weights"], laws))
        for k in range(9)
    ]


def transition_terms(m):
    previous = m["previous"]
    if previous is None:
        return None
    old_map, sources, center, step, opened, old_seen = previous
    if not sources or m["step"] != step + 1:
        return None
    if m["step"] % 25 == 0 or step % 25 == 0:
        return None

    terrain = dict(old_map)
    for p, cell in m["local"].items():
        old = old_map.get(p)
        if old is not None and old != cell:
            if old in (1, 3) or cell in (1, 3):
                return None
        terrain[p] = cell

    epoch = 25 * (step // 25)
    complete = all(
        add(source, d) in terrain
        and (
            add(source, d) in m["local"]
            or old_seen.get(add(source, d), -1) >= epoch
        )
        for source in sources for d in MOVES
    )
    # The old central square has no unobserved possible predecessors.
    monitored = (
        set(m["local"]) if complete else
        {p for p in m["local"] if distance(p, center) <= 1}
    )
    required = {
        p for p in m["enemies"] if distance(p, center) <= 1
    }
    options = []
    for source in sources:
        groups = {}
        for k, d in enumerate(MOVES):
            attempted = add(source, d)
            if (not complete and source not in monitored
                    and attempted not in monitored):
                label = None
            else:
                q = destination(
                    terrain if complete else old_map, source, d, opened
                )
                if q in monitored:
                    if q not in m["enemies"]:
                        continue
                    label = q
                else:
                    label = None
            groups[label] = groups.get(label, 0) | (1 << k)
        if not groups:
            return None
        options.append(tuple(groups.items()))

    terms = []

    def visit(index, covered, masks):
        if index == len(options):
            if required.issubset(covered):
                terms.append(tuple(masks))
            return
        for label, mask in options[index]:
            visit(
                index + 1,
                covered if label is None else covered | {label},
                masks + [mask],
            )

    visit(0, set(), [])
    return terms or None


def evidence_and_counts(law, terms):
    masks = {mask for term in terms for mask in term}
    probabilities = {
        mask: sum(law[k] for k in range(9) if mask & (1 << k))
        for mask in masks
    }
    evidence = 0.0
    posterior = [[0.0] * 9 for _ in terms[0]]
    for term in terms:
        weight = 1.0
        for mask in term:
            weight *= probabilities[mask]
        evidence += weight
        for row, mask in zip(posterior, term):
            denominator = probabilities[mask]
            if denominator > 1e-15:
                for k in range(9):
                    if mask & (1 << k):
                        row[k] += weight * law[k] / denominator

    expected = [0.0] * 9
    if evidence > 1e-15:
        for row in posterior:
            row = [v / evidence for v in row]
            information = min(
                1.0, sum(abs(a - b) for a, b in zip(row, law))
            )
            for k in range(9):
                expected[k] += information * row[k]
    return evidence, expected


def directional_information(terms):
    posterior = [[0.0] * 9 for _ in terms[0]]
    evidence = 0.0
    for term in terms:
        sizes = [mask.bit_count() for mask in term]
        weight = 1.0
        for size in sizes:
            weight *= size / 9.0
        evidence += weight
        for row, mask, size in zip(posterior, term, sizes):
            if size:
                for k in range(9):
                    if mask & (1 << k):
                        row[k] += weight / size
    if evidence <= 1e-15:
        return 0.0
    information = 0.0
    for row in posterior:
        for value in row:
            probability = value / evidence
            if probability > 1e-15:
                information += probability * math.log(9.0 * probability)
    return min(
        float(len(posterior)),
        max(0.0, information / math.log(9.0)),
    )


def cap_counts(counts, cap):
    total = sum(counts)
    if total > cap:
        for k in range(9):
            counts[k] *= cap / total


def rebuild_fast(m):
    counts = [0.0] * 9
    recent = m["recent_terms"]
    for _ in range(3):
        law = count_law(counts)
        revised = [0.0] * 9
        for index, (_, terms) in enumerate(recent):
            evidence, expected = evidence_and_counts(law, terms)
            if evidence <= 1e-15:
                continue
            recency = .78 ** (len(recent) - 1 - index)
            for k in range(9):
                revised[k] += recency * expected[k]
        cap_counts(revised, 10.0)
        counts = revised

    if len(m["counts"]) == 2:
        m["counts"].append(counts)
        retained = list(m["weights"])
    else:
        m["counts"][2] = counts
        retained = list(m["weights"][:3])
    total = sum(retained)
    m["weights"] = [.88 * w / total for w in retained] + [.12]
    m["surprise"] = [0.0, 0.0]
    m["surprise_signs"] = []
    m["last_reset"] = m["step"]
    m["recoveries"] += 1


def update_dynamics(m, obs):
    m["directional_information"] = 0.0
    if not obs.get("learn", True):
        return

    for counts, decay in zip(m["counts"], (.992, .88, .78)):
        for k in range(9):
            counts[k] *= decay
    priors = (
        (.30, .45, .25) if len(m["weights"]) == 3
        else (.30, .40, .20, .10)
    )
    m["weights"] = [
        .975 * w + .025 * prior
        for w, prior in zip(m["weights"], priors)
    ]
    m["surprise"] = [.97 * v for v in m["surprise"]]
    m["surprise_signs"] = [
        item for item in m["surprise_signs"]
        if m["step"] - item[0] <= 16
    ]
    m["recent_terms"] = [
        item for item in m["recent_terms"]
        if m["step"] - item[0] <= 16
    ]

    terms = transition_terms(m)
    if not terms:
        return
    results = [
        evidence_and_counts(law, terms) for law in component_laws(m)
    ]
    if results[0][0] >= 1.0 - 1e-10:
        return

    information = directional_information(terms)
    m["directional_information"] = information
    m["information_total"] += information
    count = max(1, len(terms[0]))
    contamination = .004 * (1.0 / 9.0) ** count
    likelihood = [
        max(1e-30, .996 * result[0] + contamination)
        for result in results
    ]

    if information >= .08:
        resolution = information / count
        advantages = [
            resolution * math.log(likelihood[0] / likelihood[1]),
            resolution * math.log(likelihood[2] / likelihood[1]),
        ]
        allowance = .035 * information
        for i, advantage in enumerate(advantages):
            m["surprise"][i] = max(
                0.0,
                m["surprise"][i]
                + max(-2.0, min(2.0, advantage)) - allowance,
            )
        m["surprise_signs"].append((
            m["step"],
            tuple(a > allowance for a in advantages),
        ))
        m["surprise_signs"] = m["surprise_signs"][-5:]
        m["recent_terms"] = (
            m["recent_terms"] + [(m["step"], terms)]
        )[-5:]

    weights = [
        w * probability ** .8
        for w, probability in zip(m["weights"], likelihood)
    ]
    total = sum(weights)
    if total > 1e-25:
        floor = .06 / len(weights)
        m["weights"] = [.94 * w / total + floor for w in weights]

    for counts, result, cap in zip(
            m["counts"], results[1:], (80.0, 16.0, 10.0)):
        if result[0] > 1e-15:
            for k in range(9):
                counts[k] += result[1][k]
            cap_counts(counts, cap)

    sustained = (
        information >= .08
        and m["surprise"][1] > 1.15
        and sum(signs[1] for _, signs in m["surprise_signs"]) >= 3
    )
    if (sustained and sum(m["counts"][0]) >= 10.0
            and len(m["recent_terms"]) >= 3
            and m["step"] - m["last_reset"] >= 10):
        rebuild_fast(m)
    m["updates"] += 1


def make_transition(m, law):
    terrain = m["map"]
    opened = opening_now(m)
    left, right, top, bottom = m["enemy_bounds"]
    cache = {}

    def transition(p):
        if p not in cache:
            row = {}
            for d, probability in zip(MOVES, law):
                q = destination(terrain, p, d, opened)
                if not (left <= q[0] <= right and top <= q[1] <= bottom):
                    q = p
                row[q] = row.get(q, 0.0) + probability
            cache[p] = row
        return cache[p]

    return transition


def propagate(distribution, transition):
    result = {}
    for p, mass in distribution.items():
        if mass < 1e-10:
            continue
        for q, probability in transition(p).items():
            result[q] = result.get(q, 0.0) + mass * probability
    return result


def occupancy(tracks, fog, p):
    empty = math.exp(-max(0.0, fog.get(p, 0.0)))
    for track in tracks:
        empty *= 1.0 - clamp(track.get(p, 0.0))
    return clamp(1.0 - empty)


def reconcile_fog(m):
    terrain = m["map"]
    physical = [p for p, cell in terrain.items() if cell != -1]
    left = max(-13, max(p[0] for p in physical) - 13)
    right = min(13, min(p[0] for p in physical) + 13)
    top = max(-13, max(p[1] for p in physical) - 13)
    bottom = min(13, min(p[1] for p in physical) + 13)

    for p, cell in terrain.items():
        if cell != -1:
            continue
        for dx, dy in CARDINAL:
            q = add(p, (dx, dy))
            if q not in terrain or terrain[q] == -1:
                continue
            if dx == 1:
                left = max(left, q[0] + 1)
            elif dx == -1:
                right = min(right, q[0] - 1)
            elif dy == 1:
                top = max(top, q[1] + 1)
            else:
                bottom = min(bottom, q[1] - 1)

    m["enemy_bounds"] = (left, right, top, bottom)
    support = {}
    for y in range(top, bottom + 1):
        for x in range(left, right + 1):
            p = (x, y)
            if p not in m["local"] and not blocked(terrain, p, m["door_open"]):
                support[p] = 1.0 if p in terrain else .68

    missing = max(0, 3 - len(m["enemies"]))
    if not missing or not support:
        return {}
    prior_total = sum(support.values())
    residual = {
        p: max(0.0, m["density"].get(p, 0.0)) for p in support
    }
    residual_total = sum(residual.values())
    if residual_total <= 1e-12:
        return {p: missing * w / prior_total for p, w in support.items()}
    return {
        p: missing * (
            .995 * residual[p] / residual_total
            + .005 * w / prior_total
        )
        for p, w in support.items()
    }


def first_forecast(m, law):
    transition = make_transition(m, law)
    tracks = [dict(transition(p)) for p in sorted(m["enemies"])]
    return tracks, propagate(m["fog"], transition), transition


def world_model_step(memory, local_obs, last_action):
    if memory is None:
        memory = {
            "map": {}, "seen": {}, "pos": (0, 0), "visits": {},
            "counts": [[0.0] * 9, [0.0] * 9],
            "weights": [.30, .45, .25], "updates": 0,
            "previous": None, "target": None, "density": {},
            "fog": {}, "stall": 0, "last_distance": None,
            "last_objective": None, "surprise": [0.0, 0.0],
            "surprise_signs": [], "recent_terms": [],
            "last_reset": -100, "recoveries": 0,
            "route_background": 0.0,
            "directional_information": 0.0, "information_total": 0.0,
            "continuation_target": None, "recent": [],
        }
    m = memory
    m["pos"] = add(m["pos"], local_obs["feedback"]["displacement"])
    m["keys"] = int(local_obs["keys"])
    m["door_open"] = bool(local_obs["door_open"])
    m["step"] = int(local_obs["step"])
    m["visits"][m["pos"]] = m["visits"].get(m["pos"], 0) + 1
    m["recent"] = (m["recent"] + [m["pos"]])[-12:]

    local, enemies = {}, set()
    for y, row in enumerate(local_obs["terrain"]):
        for x, cell in enumerate(row):
            p = add(m["pos"], (x - 2, y - 2))
            local[p] = int(cell)
            m["map"][p] = int(cell)
            m["seen"][p] = m["step"]
            if local_obs["grid"][y][x] == 5:
                enemies.add(p)
    m["local"], m["enemies"] = local, enemies

    update_dynamics(m, local_obs)
    m["law"] = learned_law(m)
    m["fog"] = reconcile_fog(m)
    forecasts, density = [], {}
    cells = set(m["map"])
    for weight, law in zip(m["weights"], component_laws(m)):
        tracks, fog, _ = first_forecast(m, law)
        forecasts.append((list(law), tracks, fog))
        for distribution in [fog] + tracks:
            cells.update(distribution)
            for p, mass in distribution.items():
                density[p] = density.get(p, 0.0) + weight * mass

    m["forecast_components"] = forecasts
    m["prediction"] = {
        p: 0.0 if blocked(m["map"], p, opening_now(m)) else clamp(sum(
            weight * occupancy(tracks, fog, p)
            for weight, (_, tracks, fog) in zip(m["weights"], forecasts)
        ))
        for p in cells
    }
    m["default"] = 0.0
    m["density"] = density
    m["previous"] = (
        dict(m["map"]), tuple(sorted(enemies)), m["pos"],
        m["step"], opening_now(m), dict(m["seen"]),
    )
    return m


def legal(m, p, q, immediate=False):
    dx, dy = q[0] - p[0], q[1] - p[1]
    opened = opening_now(m) if immediate else (
        m["door_open"] or m["keys"] >= 2
    )
    if blocked(m["map"], q, opened, True):
        return False
    if dx and dy:
        if (blocked(m["map"], (p[0] + dx, p[1]), opened, True)
                or blocked(m["map"], (p[0], p[1] + dy), opened, True)):
            return False
        if m["map"].get(q) == 3 and not m["door_open"]:
            return False
    return True


def graph(m):
    opened = m["door_open"] or m["keys"] >= 2
    return {
        p: [add(p, d) for d in WALK if legal(m, p, add(p, d))]
        for p in m["map"] if not blocked(m["map"], p, opened, True)
    }


def reverse_graph(edges):
    result = {p: [] for p in edges}
    for p, neighbors in edges.items():
        for q in neighbors:
            result.setdefault(q, []).append(p)
    return result


def breadth_distances(edges, source):
    result, queue = {source: 0}, [source]
    for p in queue:
        for q in edges.get(p, ()):
            if q not in result:
                result[q] = result[p] + 1
                queue.append(q)
    return result


def paths(edges, source, m, risk, reverse=False):
    adjacency = reverse_graph(edges) if reverse else edges
    result, queue = {source: 0.0}, [(0.0, source)]
    while queue:
        cost, p = heapq.heappop(queue)
        if cost != result[p]:
            continue
        for q in adjacency.get(p, ()):
            cell = p if reverse else q
            new = (
                cost + 1.0 + 4.0 * risk.get(cell, 0.0)
                + .02 * min(15, m["visits"].get(cell, 0))
            )
            if new < result.get(q, float("inf")):
                result[q] = new
                heapq.heappush(queue, (new, q))
    return result


def hazard(p):
    return -math.log(max(1e-10, 1.0 - clamp(p)))


def mixture_hazard(probabilities, weights):
    total = sum(weights)
    if total <= 1e-15:
        return hazard(1.0)
    weights = [w / total for w in weights]
    mean = sum(w * p for w, p in zip(weights, probabilities))
    cutoff = max(.08, .20 * max(weights))
    plausible = [
        (w, p) for w, p in zip(weights, probabilities) if w >= cutoff
    ]
    mass = sum(w for w, _ in plausible)
    center = sum(w * p for w, p in plausible) / mass
    variance = sum(w * (p - center) ** 2 for w, p in plausible) / mass
    upper = max(mean, max(p for _, p in plausible))
    return hazard(min(upper, mean + .45 * math.sqrt(max(0.0, variance))))


def prepare_second(expert):
    tracks, fog, transition = expert
    return (
        tracks, [propagate(track, transition) for track in tracks],
        fog, propagate(fog, transition), transition,
    )


def second_probability(state, q, r):
    tracks, after, fog, fog_after, transition = state
    qr = transition(q).get(r, 0.0)
    rr = transition(r).get(r, 0.0)
    total = 0.0
    for old, new in zip(tracks, after):
        at_q = old.get(q, 0.0)
        surviving = 1.0 - at_q
        if surviving <= 1e-12:
            return 1.0
        danger = max(0.0, new.get(r, 0.0) - at_q * qr) / surviving
        if r != q:
            danger += old.get(r, 0.0) * (1.0 - rr) / surviving
        total += hazard(danger)
    intensity = max(0.0, fog_after.get(r, 0.0) - fog.get(q, 0.0) * qr)
    if r != q:
        intensity += fog.get(r, 0.0) * (1.0 - rr)
    return clamp(-math.expm1(-total - max(0.0, intensity)))


def third_probabilities(prepared, q, r, cells):
    """Condition each enemy on survival along both preceding actions."""
    needed = {add(s, d) for s in cells for d in MOVES}
    rows = [[] for _ in cells]
    excluded = (q,) if q == r else (q, r)
    for tracks, after, fog, fog_after, transition in prepared:
        kernels = {p: transition(p) for p in excluded}
        conditioned = []
        impossible = False
        for old, new in zip(tracks, after):
            removed = [(p, old.get(p, 0.0)) for p in excluded]
            killed_after = max(
                0.0,
                new.get(r, 0.0)
                - sum(mass * kernels[p].get(r, 0.0) for p, mass in removed),
            )
            surviving = 1.0 - sum(mass for _, mass in removed) - killed_after
            if surviving <= 1e-12:
                impossible = True
                break
            conditioned.append({
                s: 0.0 if s == r else clamp(
                    max(0.0, new.get(s, 0.0) - sum(
                        mass * kernels[p].get(s, 0.0)
                        for p, mass in removed
                    )) / surviving
                )
                for s in needed
            })
        if impossible:
            for row in rows:
                row.append(1.0)
            continue

        remaining_fog = {
            s: 0.0 if s == r else max(
                0.0,
                fog_after.get(s, 0.0)
                - sum(
                    fog.get(p, 0.0) * kernels[p].get(s, 0.0)
                    for p in excluded
                ),
            )
            for s in needed
        }
        for index, s in enumerate(cells):
            incoming = [
                (p, transition(p).get(s, 0.0))
                for p in (add(s, d) for d in WALK)
                if remaining_fog.get(p, 0.0) > 0.0
                or any(track.get(p, 0.0) > 0.0 for track in conditioned)
            ]
            total = remaining_fog.get(s, 0.0) + sum(
                remaining_fog.get(p, 0.0) * probability
                for p, probability in incoming
            )
            for track in conditioned:
                total += hazard(
                    track.get(s, 0.0) + sum(
                        track.get(p, 0.0) * probability
                        for p, probability in incoming
                    )
                )
            rows[index].append(clamp(-math.expm1(-total)))
    return rows


def route_forecasts(m, experts, prepared, weights, law):
    cells = list(m["map"])
    fields = [
        {
            p: sum(
                w * occupancy(tracks, fog, p)
                for w, (tracks, fog, _) in zip(weights, experts)
            )
            for p in cells
        },
        {
            p: sum(
                w * occupancy(state[1], state[3], p)
                for w, state in zip(weights, prepared)
            )
            for p in cells
        },
    ]
    tracks = [{} for _ in experts[0][0]]
    fog = {}
    for weight, state in zip(weights, prepared):
        for pooled, distribution in zip(tracks, state[1]):
            for p, mass in distribution.items():
                pooled[p] = pooled.get(p, 0.0) + weight * mass
        for p, mass in state[3].items():
            fog[p] = fog.get(p, 0.0) + weight * mass

    transition = make_transition(m, law)
    for _ in range(2, ROUTE_HORIZON):
        tracks = [propagate(track, transition) for track in tracks]
        fog = propagate(fog, transition)
        fields.append({p: occupancy(tracks, fog, p) for p in cells})

    left, right, top, bottom = m["enemy_bounds"]
    area = sum(
        .68 if (x, y) not in m["map"] else
        float(not blocked(m["map"], (x, y), opening_now(m)))
        for y in range(top, bottom + 1)
        for x in range(left, right + 1)
    )
    background = min(.06, max(.012, 3.0 / max(60.0, area)))
    m["route_background"] = background
    persistence = {p: transition(p).get(p, 0.0) for p in cells}
    return fields, persistence, background


def arrival_risk(edges, source, forecast, offset=0):
    fields, persistence, background = forecast
    risk = {}
    for p, steps in breadth_distances(edges, source).items():
        arrival = max(1, int(offset) + steps)
        if arrival <= ROUTE_HORIZON:
            value = fields[arrival - 1].get(p, background)
        else:
            retained = math.exp(
                -(arrival - ROUTE_HORIZON)
                / (3.0 + 5.0 * persistence.get(p, 0.0))
            )
            value = (
                retained * fields[-1].get(p, background)
                + (1.0 - retained) * background
            )
        risk[p] = clamp(value)
    return risk


def view_gain(m, p, exclude=None):
    result = 0
    for dy in range(-2, 3):
        for dx in range(-2, 3):
            q = add(p, (dx, dy))
            if q not in m["map"]:
                if exclude is None or distance(q, exclude) > 2:
                    result += 1
    return result


def choose_target(m, edges, dist, forecast):
    start = m["pos"]
    desired = 2 if m["keys"] < 2 else 4
    goals = [
        p for p in dist if p != start and m["map"].get(p) == desired
    ]
    if goals:
        if desired == 2 and m["keys"] == 0 and len(goals) > 1:
            doors = [p for p, cell in m["map"].items() if cell in (3, 4)]
            arrivals = breadth_distances(edges, start)
            tours = []
            for first in goals:
                risk = arrival_risk(
                    edges, first, forecast, arrivals.get(first, 0)
                )
                onward = paths(edges, first, m, risk)
                for second in goals:
                    if second == first or second not in onward:
                        continue
                    tail = min((distance(second, p) for p in doors), default=0)
                    tours.append((
                        dist[first] + onward[second] + .6 * tail, first
                    ))
            if tours:
                return min(tours)[1]
        return min(goals, key=lambda p: (dist[p], p))

    if m["keys"] >= 2 and not m["door_open"]:
        doors = [
            p for p in dist if p != start and m["map"].get(p) == 3
        ]
        if doors:
            return min(doors, key=lambda p: (dist[p], p))

    frontier, fallback = [], []
    tick = 25 * (m["step"] // 25)
    for p, travel in dist.items():
        if p == start:
            continue
        unseen = view_gain(m, p)
        stale = sum(
            m["map"].get(add(p, (dx, dy))) not in (None, -1)
            and m["seen"].get(add(p, (dx, dy)), m["step"]) < tick
            for dy in range(-2, 3) for dx in range(-2, 3)
        )
        visits = m["visits"].get(p, 0)
        if unseen:
            score = (travel + .15 * visits) / unseen ** .55
            if p == m["target"]:
                score *= .94
            frontier.append((score, p))
        fallback.append((travel + .9 * visits - .20 * stale, p))
    if frontier:
        return min(frontier)[1]
    return min(fallback)[1] if fallback else start


def continuation_field(m, edges, target, risk):
    """An anchored, bounded potential after completing the current target."""
    onward = paths(edges, target, m, risk)
    keys_after = min(2, m["keys"] + int(m["map"].get(target) == 2))
    desired = 2 if keys_after < 2 else 4
    goals = [
        p for p in onward
        if p != target and m["map"].get(p) == desired
    ]
    if not goals and keys_after >= 2:
        goals = [
            p for p in onward
            if p != target and m["map"].get(p) == 3
        ]
    if goals:
        successor = min(goals, key=lambda p: (onward[p], p))
    else:
        candidates = []
        for p, cost in onward.items():
            if p == target:
                continue
            gain = view_gain(m, p, exclude=target)
            if gain:
                candidates.append((
                    (cost + .2) / gain ** .55
                    + .025 * m["visits"].get(p, 0),
                    p,
                ))
        successor = min(candidates)[1] if candidates else None
    m["continuation_target"] = successor
    if successor is None:
        return {p: 0.0 for p in onward}
    potential = paths(edges, successor, m, risk, True)
    anchor = potential.get(target, 0.0)
    return {
        p: .40 * max(-4.0, min(6.0, cost - anchor))
        for p, cost in potential.items()
    }


def is_exit(m, p):
    return (
        m["map"].get(p) == 4
        and m["keys"] >= 2
        and opening_now(m)
    )


def planner(memory, local_obs):
    m = memory
    start = m["pos"]
    predictive = local_obs.get("predictive_planning", True)
    nearby = (
        bool(m["enemies"])
        or any(
            distance(start, p) <= 2 and probability > .006
            for p, probability in m["prediction"].items()
        )
    )
    law = m["law"] if predictive else UNIFORM
    if predictive and nearby:
        weights = list(m["weights"])
        experts = [
            (tracks, fog, make_transition(m, component))
            for component, tracks, fog in m["forecast_components"]
        ]
    else:
        experts, weights = [first_forecast(m, law)], [1.0]

    prepared = [prepare_second(expert) for expert in experts]
    edges = graph(m)
    forecast = route_forecasts(m, experts, prepared, weights, law)
    route_risk = arrival_risk(edges, start, forecast)
    dist = paths(edges, start, m, route_risk)
    target = choose_target(m, edges, dist, forecast)
    potential = paths(edges, target, m, route_risk, True)
    geometry = breadth_distances(reverse_graph(edges), target)

    objective = (target, m["keys"], m["door_open"])
    progress = geometry.get(start, 225)
    if objective == m["last_objective"]:
        if progress < m["last_distance"]:
            m["stall"] = max(0, m["stall"] - 3)
        else:
            m["stall"] += 1
    else:
        m["stall"] = max(0, m["stall"] - 3)
    m["last_objective"] = objective
    m["last_distance"] = progress
    m["target"] = target

    remaining = max(1, 200 - m["step"])
    slack = remaining - potential.get(start, 25.0)
    pressure = min(92.0, max(24.0, 1.8 * slack))
    pressure /= 1.0 + .035 * max(0, m["stall"] - 8)
    pressure = max(22.0, pressure)
    waiting = .18 + .03 * min(16, m["stall"])

    # Only calculate a successor when target completion is in the local horizon.
    if geometry.get(start, 225) <= 3 and not is_exit(m, target):
        onward = continuation_field(m, edges, target, route_risk)
    else:
        onward = {}
        m["continuation_target"] = None
    completion_reward = 1.4 if m["map"].get(target) in (2, 3) else .9

    def complete(p, reached):
        return reached or (target != start and p == target)

    def reward(p, reached, tick):
        if not reached and target != start and p == target:
            return completion_reward / (1.0 + .15 * (tick - 1))
        return 0.0

    def tail(p, reached):
        if reached:
            return onward.get(p, 0.0)
        return potential.get(p, 250.0)

    def movement_cost(p, q):
        return (
            1.0 + (waiting if p == q else 0.0)
            + .025 * min(16, m["visits"].get(q, 0))
        )

    choices = []
    third_cache = {}
    for d in MOVES:
        q = add(start, d)
        if (not legal(m, start, q, True)
                or q in m["enemies"] or q not in potential):
            continue

        probabilities = [
            occupancy(tracks, fog, q) for tracks, fog, _ in experts
        ]
        h1 = mixture_hazard(probabilities, weights)
        score = movement_cost(start, q) + pressure * h1
        score -= reward(q, False, 1)
        if is_exit(m, q):
            choices.append((score, h1, d))
            continue

        reached_q = complete(q, False)
        survivor_weights = [
            w * max(0.0, 1.0 - p)
            for w, p in zip(weights, probabilities)
        ]
        second_choices = []
        for e in MOVES:
            r = add(q, e)
            if r not in potential or not legal(m, q, r):
                continue
            probabilities2 = [
                second_probability(state, q, r) for state in prepared
            ]
            h2 = mixture_hazard(probabilities2, survivor_weights)
            local_cost = movement_cost(q, r) + .58 * pressure * h2
            local_cost -= reward(r, reached_q, 2)
            reached_r = complete(r, reached_q)
            proxy = local_cost + (0.0 if is_exit(m, r) else tail(r, reached_r))
            second_choices.append((
                proxy, r, local_cost, probabilities2, reached_r,
            ))

        continuation = float("inf")
        for _, r, local_cost, probabilities2, reached_r in sorted(
                second_choices)[:SECOND_BEAM]:
            if is_exit(m, r):
                continuation = min(continuation, local_cost)
                continue
            surviving = [
                w * max(0.0, 1.0 - p)
                for w, p in zip(survivor_weights, probabilities2)
            ]
            cache_key = q, r
            if cache_key not in third_cache:
                cells = [
                    add(r, e) for e in MOVES
                    if add(r, e) in potential and legal(m, r, add(r, e))
                ]
                third_cache[cache_key] = (
                    cells, third_probabilities(prepared, q, r, cells)
                )
            cells, rows = third_cache[cache_key]
            third_best = float("inf")
            for s, row in zip(cells, rows):
                reached_s = complete(s, reached_r)
                value = (
                    movement_cost(r, s)
                    + .27 * pressure * mixture_hazard(row, surviving)
                    - reward(s, reached_r, 3)
                )
                if not is_exit(m, s):
                    value += tail(s, reached_s)
                third_best = min(third_best, value)
            if math.isfinite(third_best):
                continuation = min(continuation, local_cost + third_best)

        score += (
            continuation if math.isfinite(continuation)
            else tail(q, reached_q) + pressure
        )
        choices.append((score, h1, d))

    if choices:
        move = min(choices)[2]
    else:
        emergency = []
        for d in MOVES:
            q = add(start, d)
            if legal(m, start, q, True) and q not in m["enemies"]:
                emergency.append((
                    mixture_hazard(
                        [occupancy(tracks, fog, q) for tracks, fog, _ in experts],
                        weights,
                    ),
                    potential.get(q, 1e6),
                    d,
                ))
        move = min(emergency)[2] if emergency else (0, 0)
    return {"move": [int(move[0]), int(move[1])], "interact": True}
