<DESCRIPTION>
Replace independent before/after collision estimates with per-enemy survival conditioning. Each hypothetical move removes trajectories that collide before movement or after movement, preserving the remaining distribution for subsequent decisions. This avoids double-counting persistent enemies and exposes routes that become dangerous after an initially safe move.

Extend planning to three steps, refining the three most promising second moves for each immediate action to limit computation. Retain the existing learner, mapping, exploration, export predictions, and uniform-dynamics intervention. Hidden-enemy risk remains an approximate background term. This is an untested control improvement; higher task fitness requires evaluation.
</DESCRIPTION>

            for d in ((1, 0), (-1, 0), (0, 1), (0, -1))))

        if p not in cache:

                q = _destination(m["map"], p, d, opened)
                result[q] = result.get(q, 0.0) + probability
            cache[p] = result
        return cache[p]
    return transition
def _project_tracks(tracks, transition):
    projected = []
    for track in tracks:
        result = {}
        for p, mass in track.items():
            for q, probability in transition(p).items():
                result[q] = result.get(q, 0.0) + mass * probability
        projected.append(result)
    return projected
def _collision_cost(tracks, projected, q, transition):
    # For each enemy, collision is the union of occupancy at q before
    # movement and occupancy there afterwards. These events overlap.
    stay = transition(q).get(q, 0.0)
    result = 0.0
    for before, after in zip(tracks, projected):
        danger = after.get(q, 0.0) + before.get(q, 0.0) * (1.0 - stay)
        if danger >= 1.0 - 1e-12:
            return float("inf")
        result += _hazard(danger)
    return result
def _surviving_tracks(tracks, projected, q, transition):
    # Remove trajectories starting at q, then trajectories ending at q.
    # The resulting tracks are conditional on surviving the entire action.
    outgoing = transition(q)
    result = []
    for before, after in zip(tracks, projected):
        at_q = before.get(q, 0.0)
        filtered = {}
        for p, mass in after.items():
            if p != q:
                surviving = mass - at_q * outgoing.get(p, 0.0)
                if surviving > 0.0:
                    filtered[p] = surviving
        total = sum(filtered.values())
        if total <= 1e-12:
            return None
        result.append({p: mass / total for p, mass in filtered.items()})
    return result
def _background_cost(m, p, q, depth):
    def probability(cell, horizon):
        if horizon <= 0:
            return 0.0
        if _distance(cell, m["pos"]) <= 2 - horizon:
            return 0.0
        if cell in m["local"]:
            return m["background"] * min(1.0, .18 * horizon)
        return m["background"]
    after = probability(q, depth)
    before = 0.0 if p == q else probability(q, depth - 1)
    return _hazard(before) + _hazard(after)
