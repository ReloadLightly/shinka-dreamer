<DESCRIPTION>
Carry the door’s opening time through the existing three-step planner. Each branch opens it only when interaction occurs from a cardinal neighbor with two previously collected keys. Enemy forecasts then use closed-door kernels before that time and open-door kernels afterward, preserving blocked-attempt probability mass. Memoization includes the opening schedule and hypothetical key pickups.

Also make immediate forecasts and learning evidence respect actual interaction timing. This targets inconsistent door geometry without changing the learner’s adaptation rates, risk penalties, exploration strategy, or planning horizon. Improved control is a hypothesis; this patch has not been evaluated.
</DESCRIPTION>

def _interaction_open(m, p, opened=None, keys=None):
    if opened is None:
        opened = m["door_open"]
    if keys is None:
        keys = m["keys"]
    return bool(opened or (
        keys >= 2 and any(
            m["map"].get(_add(p, d)) == 3
            for d in ((1, 0), (-1, 0), (0, 1), (0, -1))
        )
    ))


def _blocked(m, p, opened=None):
    cell = m["map"].get(p, 1)
    if cell in (-1, 1):
        return True
    # Without a branch state, retain an optimistic routing heuristic.
    # Actual action legality always supplies the post-interaction state.
    if opened is None:
        opened = m["door_open"] or m["keys"] >= 2
    return cell == 3 and not opened


def _legal(m, p, d, opened=None):
    if opened is None:
        opened = _interaction_open(m, p)
    q = _add(p, d)
    if _blocked(m, q, opened):
        return False
    if d[0] and d[1]:
        if (_blocked(m, _add(p, (d[0], 0)), opened)
                or _blocked(m, _add(p, (0, d[1])), opened)):
            return False
    return True
